```bash
// usage: node coverage-gaps.mjs <worktree> <src/relative/path>...
// prints, per file, uncovered statements / functions / branches after a coverage run.
import { readFileSync } from 'node:fs';
import { join, resolve } from 'node:path';
const [wt, ...files] = process.argv.slice(2);
const data = JSON.parse(readFileSync(join(wt, 'coverage', 'coverage-final.json'), 'utf8'));
for (const f of files) {
  const abs = resolve(wt, f).replace(/\\/g, '/');
  const entry = Object.entries(data).find(([k]) => k.replace(/\\/g, '/') === abs)?.[1];
  if (!entry) { console.log(`${f}: not in coverage`); continue; }
  const s = Object.entries(entry.s).filter(([, n]) => n === 0).map(([k]) => entry.statementMap[k].start.line);
  const fn = Object.entries(entry.f).filter(([, n]) => n === 0).map(([k]) => `${entry.fnMap[k].name}@${entry.fnMap[k].loc.start.line}`);
  const b = Object.entries(entry.b).flatMap(([k, arr]) => arr.map((n, i) => n === 0 ? `${entry.branchMap[k].loc.start.line}[${i}]` : null).filter(Boolean));
  console.log(`${f}\n  statements: ${s.join(',') || '-'}\n  functions: ${fn.join(',') || '-'}\n  branches: ${b.join(',') || '-'}`);
}
JS
```

```ts
import { expect } from 'vitest';
import type Konva from 'konva';

import { STAGE_PIXELS, worldToScreen } from '../../src/presentation/editor/viewport/Viewport';

/** Preserve exact vertex counts and exercise the real idle hover before expecting rotation arrows. */
export async function zoneEditingHandles(harness: EditorHarness, pointCount: number, hoverWorld: Point) {
	const interaction = expectDefined(harness.stage?.findOne<Konva.Layer>('.interaction'), 'interaction layer');
	expect(interaction.getChildren().filter(node => node.getClassName() === 'Circle')).toHaveLength(pointCount);

	const hover = worldToScreen(hoverWorld, useEditorStore(harness.pinia).viewport, STAGE_PIXELS);
	pointer(canvas, 'pointermove', box.left + hover.x, box.top + hover.y, 0, 1, 0); await settle();
	expect(interaction.findOne('.object-rotation-handle')).toBeDefined();
	return interaction;
}
```
